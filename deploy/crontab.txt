# Plain-cron alternative to deploy/systemd/ (use one or the other, never both).
# Install as the tradingbot user:   crontab deploy/crontab.txt
#
# Times are UTC (the first line forces it; Debian/Ubuntu cron honours CRON_TZ
# only on some builds, so the times below are chosen to need no time zone at all).
# NYSE is 13:30-20:00 UTC under US daylight time and 14:30-21:00 UTC under
# standard time; 14:45 and 17:30 UTC are inside the session under both, so these
# never need a seasonal edit. (systemd timers can pin New York time directly --
# prefer them if you have systemd.)
CRON_TZ=UTC
SHELL=/bin/bash
REPO=/opt/tradingbot

# Trading cycle: 10:45/09:45 ET, 13:30/12:30 ET, and a crypto-only overnight run.
45 14 * * *   bash $REPO/deploy/run_job.sh cycle     >> $REPO/logs/cycle.log 2>&1
30 17 * * *   bash $REPO/deploy/run_job.sh cycle     >> $REPO/logs/cycle.log 2>&1
0 2 * * *     bash $REPO/deploy/run_job.sh cycle     >> $REPO/logs/cycle.log 2>&1

# Volume watch and position refresh, every 15 min (refresh offset by 5).
*/15 * * * *  bash $REPO/deploy/run_job.sh watch     >> $REPO/logs/watch.log 2>&1
5-59/15 * * * * bash $REPO/deploy/run_job.sh refresh >> $REPO/logs/refresh.log 2>&1

# Weekly screen, Monday 06:00 UTC.
0 6 * * 1     bash $REPO/deploy/run_job.sh screening >> $REPO/logs/screening.log 2>&1
